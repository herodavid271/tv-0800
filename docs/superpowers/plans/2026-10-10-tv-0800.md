**Goal:** Página estática para los televisores de 0800 que muestra, animada, el calendario y la próxima edición de cada curso presencial leyendo en vivo del Supabase de 0800 HUB.

**Architecture:** HTML/CSS/JS sin framework ni build, servido por GitHub Pages (repo `tv-0800`). **Tech Stack:** HTML5, CSS (solo `transform`/`opacity` animados), JavaScript ES modules, Node ≥ 18 (`node:test`) solo para pruebas, PostgreSQL/PostgREST (Supabase), GitHub Pages.

- [ ] **Step 1: Escribir sql/001_tv_ediciones.sql**

```sql
create or replace function public.tv_ediciones()
returns table (
  course_slug text,
  course_name text,
  start_date date,
  end_date date,
  city text,
  location text,
  total_seats int,
  available_seats int,
  edition_status text
)
language sql
stable
security definer
set search_path = public
as $$
  select
    c.slug,
    c.name,
    e.start_date::date,
    e.end_date::date,
    e.city,
    e.location,
    e.total_seats,
    e.available_seats,
    e.status
  from public.vta_course_editions e
  join public.vta_courses c on c.id = e.course_id
  where c.course_type = 'in_person'
    and c.status = 'active'
    and c.requires_edition
    and e.status in ('active', 'finished')
    and (
      (e.status = 'active' and e.end_date::date >= current_date - 1)
      or e.start_date = (
        select max(e2.start_date)
        from public.vta_course_editions e2
        where e2.course_id = c.id
          and e2.status in ('active', 'finished')
      )
    )
  order by e.start_date, c.name;
$$;

revoke all on function public.tv_ediciones() from public, anon, authenticated;
grant execute on function public.tv_ediciones() to anon, authenticated;
```

```js
export const MESES = ['enero', 'febrero', 'marzo', 'abril', 'mayo', 'junio', 'julio', 'agosto', 'septiembre', 'octubre', 'noviembre', 'diciembre'];
export const DIAS = ['domingo', 'lunes', 'martes', 'miércoles', 'jueves', 'viernes', 'sábado'];

const pad = (n) => String(n).padStart(2, '0');

export function localISO(d = new Date()) {
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
}

export function parseISO(iso) {
  const [y, m, d] = iso.split('-').map(Number);
  return { y, m, d };
}

const utc = (iso) => {
  const { y, m, d } = parseISO(iso);
  return Date.UTC(y, m - 1, d);
};

export function daysBetween(fromISO, toISO) {
  return Math.round((utc(toISO) - utc(fromISO)) / 86400000);
}

export function addDaysISO(iso, n) {
  const dt = new Date(utc(iso) + n * 86400000);
  return `${dt.getUTCFullYear()}-${pad(dt.getUTCMonth() + 1)}-${pad(dt.getUTCDate())}`;
}

export function formatLong(iso) {
  const { y, m, d } = parseISO(iso);
  return `${DIAS[new Date(Date.UTC(y, m - 1, d)).getUTCDay()]} ${d} de ${MESES[m - 1]}`;
}

export function formatShort(iso) {
  const { m, d } = parseISO(iso);
  return `${d} ${MESES[m - 1].slice(0, 3)}`;
}
```

```js
import { daysBetween } from './dates.js';

export const LOW_SEATS = 10;

export function daysUntil(startISO, todayISO) {
  return daysBetween(todayISO, startISO);
}

export function badgeFor(days) {
  if (days < 0) return 'EN CURSO';
  if (days === 0) return 'HOY';
  if (days === 1) return 'MAÑANA';
  return null;
}

export function seatState(course) {
  if (course.availableSeats <= 0) return 'agotado';
  if (course.availableSeats <= LOW_SEATS) return 'pocos';
  return 'ok';
}

export function buildScreens(courses, config, todayISO) {
  const visible = courses.filter((c) => !config.hidden_courses.includes(c.slug));
  const upcoming = visible.filter((c) => !c.finished);
  if (upcoming.length === 0) return [{ type: 'vacio' }];

  const soon = upcoming.filter((c) => daysUntil(c.startDate, todayISO) <= 1);
  const later = upcoming.filter((c) => daysUntil(c.startDate, todayISO) > 1);
  const ordered = [...soon, ...later, ...visible.filter((c) => c.finished)];

  const screens = [];
  for (const name of config.screens) {
    if (name === 'calendario') screens.push({ type: 'calendario' });
    else if (name === 'cursos') ordered.forEach((course) => screens.push({ type: 'curso', course }));
    else if (name === 'resumen') screens.push({ type: 'resumen' });
  }
  return screens.length ? screens : [{ type: 'vacio' }];
}
```

```js
import { MESES, parseISO, addDaysISO, daysBetween } from './dates.js';

export function monthToShow(todayISO, courses) {
  const upcoming = courses.filter((c) => !c.finished).map((c) => c.startDate).sort();
  const t = parseISO(todayISO);
  const thisMonth = todayISO.slice(0, 7);
  const useToday = upcoming.length === 0 || upcoming.some((iso) => iso.startsWith(thisMonth));
  const pick = useToday ? t : parseISO(upcoming[0]);
  return { year: pick.y, month: pick.m - 1 };
}

export function buildMonthGrid(year, month, courses, todayISO) {
  const marks = new Map();
  for (const c of courses.filter((x) => !x.finished)) {
    const span = Math.max(0, daysBetween(c.startDate, c.endDate));
    for (let i = 0; i <= span; i++) {
      const iso = addDaysISO(c.startDate, i);
      if (!marks.has(iso)) marks.set(iso, []);
      marks.get(iso).push(c.slug);
    }
  }

  const pad = (n) => String(n).padStart(2, '0');
  const offset = (new Date(Date.UTC(year, month, 1)).getUTCDay() + 6) % 7;
  const daysInMonth = new Date(Date.UTC(year, month + 1, 0)).getUTCDate();

  const cells = Array(offset).fill(null);
  for (let day = 1; day <= daysInMonth; day++) {
    const iso = `${year}-${pad(month + 1)}-${pad(day)}`;
    cells.push({ day, iso, marked: marks.get(iso) ?? [], isToday: iso === todayISO });
  }
  while (cells.length % 7 !== 0) cells.push(null);

  const weeks = [];
  for (let i = 0; i < cells.length; i += 7) weeks.push(cells.slice(i, i + 7));

  return { title: `${MESES[month]} ${year}`, weekdays: ['L', 'M', 'M', 'J', 'V', 'S', 'D'], weeks };
}
```

```js
export function calendarScreen(courses, todayISO) {
  const { year, month } = monthToShow(todayISO, courses);
  const g = buildMonthGrid(year, month, courses, todayISO);
  const head = g.weekdays.map((w) => `<span class="wd">${w}</span>`).join('');
  const cells = g.weeks
    .flat()
    .map((cell, i) =>
      cell
        ? `<span class="cell${cell.marked.length ? ' marked' : ''}${cell.isToday ? ' today' : ''}" style="--i:${i}">${cell.day}</span>`
        : '<span class="cell empty"></span>')
    .join('');
  const rows = courses
    .filter((c) => !c.finished)
    .map(
      (c, i) => `<li class="row" style="--i:${i}"><span class="chip display">${parseISO(c.startDate).d}</span><div><b>${esc(c.name)}</b><small>${formatShort(c.startDate)}</small></div><span class="dleft">${daysLabel(daysUntil(c.startDate, todayISO))}</span></li>`)
    .join('');
  return `<section class="screen calendario"><div class="cal"><h1 class="display">${g.title}</h1><div class="grid">${head}${cells}</div></div><ul class="list">${rows}</ul></section>`;
}

export function courseScreen(c, todayISO) {
  const days = daysUntil(c.startDate, todayISO);
  const badge = c.finished ? 'FINALIZADO' : badgeFor(days);
  const state = seatState(c);

  const seatsKpi =
    state === 'agotado'
      ? '<div class="kpi warn"><span class="val display">AGOTADO</span></div>'
      : `<div class="kpi${state === 'pocos' ? ' warn pulse' : ''}"><span class="val display num" data-count="${c.availableSeats}">${c.availableSeats}</span><span class="lbl">cupos disponibles</span></div>`;
  const daysKpi =
    !c.finished && days > 1
      ? `<div class="kpi"><span class="val display num" data-count="${days}">${days}</span><span class="lbl">días para empezar</span></div>`
      : '';
  const kpis = c.finished ? '' : `<div class="kpis">${daysKpi}${seatsKpi}</div>`;
  const place = where(c);

  return `<section class="screen curso" data-slug="${esc(c.slug)}">
<div class="thumb"><img src="assets/cursos/${esc(c.slug)}.jpg" alt="" onerror="this.parentNode.classList.add('nothumb')"><span class="fallback display">${esc(c.name)}</span></div>
<div class="info">${badge ? `<span class="badge${c.finished ? ' done' : ''}">${badge}</span>` : ''}
<span class="eyb">Presencial</span>
<h1 class="fecha display">${formatLong(c.startDate)}</h1>
${place ? `<p class="lugar">${esc(place)}</p>` : ''}
${kpis}</div></section>`;
}

export function summaryScreen(courses, todayISO) {
  const rows = courses
    .map((c, i) => {
      const days = daysUntil(c.startDate, todayISO);
      const state = seatState(c);
      const seats = c.finished
        ? 'Finalizado'
        : state === 'agotado'
          ? '<span class="warn">AGOTADO</span>'
          : `<span class="${state === 'pocos' ? 'warn' : ''}">${c.availableSeats} cupos</span>`;
      return `<div class="tr${c.finished ? ' done' : ''}" style="--i:${i}"><b>${esc(c.name)}</b><span>${formatShort(c.startDate)}</span><span>${c.finished ? '' : daysLabel(days)}</span>${seats}</div>`;
    })
    .join('');
  return `<section class="screen resumen"><h1 class="display">Próximos cursos</h1><div class="tbl">${rows}</div></section>`;
}

export function emptyScreen(text) {
  return `<section class="screen vacio"><h1 class="display">${esc(text)}</h1></section>`;
}
```

```js
import { SUPABASE_URL, SUPABASE_KEY } from './config.js';
import { createClient, loadSnapshot, pickCourses, normalizeConfig } from './data.js';
import { buildScreens } from './rotation.js';
import { localISO, formatLong } from './dates.js';
import { calendarScreen, courseScreen, summaryScreen, emptyScreen } from './render.js';

const REFRESH_MS = 5 * 60 * 1000;
const RETRY_MS = 15 * 1000;
const $ = (id) => document.getElementById(id);
const root = document.documentElement;

const params = new URLSearchParams(location.search);
if (params.get('tema') === 'claro') root.dataset.theme = 'claro';
if (params.has('lite')) root.classList.add('lite');

const client = createClient({ url: SUPABASE_URL, key: SUPABASE_KEY });
const storage = (() => {
  try { return window.localStorage; } catch { return null; }
})();

const state = { courses: [], config: normalizeConfig(null), online: true, loaded: false, screens: [], index: 0 };
let timer = null;
let firstLoad = true;

function fit() {
  const s = Math.min(innerWidth / 1920, innerHeight / 1080);
  $('stage').style.transform = `translate(-50%, -50%) scale(${s})`;
}

function tick() {
  const now = new Date();
  $('time').textContent = now.toLocaleTimeString('es', { hour: '2-digit', minute: '2-digit', hour12: false });
  $('date').textContent = formatLong(localISO(now));
}

const visibleCourses = () => state.courses.filter((c) => !state.config.hidden_courses.includes(c.slug));

function rebuild() {
  state.screens = state.loaded
    ? buildScreens(state.courses, state.config, localISO())
    : [{ type: 'conectando' }];
  if (state.index >= state.screens.length) state.index = 0;
}

function html(screen) {
  const today = localISO();
  switch (screen.type) {
    case 'calendario': return calendarScreen(visibleCourses(), today);
    case 'curso': return courseScreen(screen.course, today);
    case 'resumen': return summaryScreen(visibleCourses(), today);
    case 'conectando': return emptyScreen('Conectando…');
    default: return emptyScreen('Próximamente');
  }
}

function animateCounts(scope) {
  if (root.classList.contains('lite') || matchMedia('(prefers-reduced-motion: reduce)').matches) return;
  scope.querySelectorAll('[data-count]').forEach((el) => {
    const to = Number(el.dataset.count);
    const t0 = performance.now();
    const D = 700;
    (function step(t) {
      const p = Math.min(1, (t - t0) / D);
      el.textContent = String(Math.round(to * (1 - Math.pow(1 - p, 3))));
      if (p < 1) requestAnimationFrame(step);
    })(t0);
  });
}

function show() {
  clearTimeout(timer);
  const screen = state.screens[state.index];
  const secs = state.config.seconds_per_screen;
  $('host').innerHTML = html(screen);
  animateCounts($('host'));
  $('progress').innerHTML = `<i style="--dur:${secs}s"></i>`;
  $('msg').textContent = state.config.message ?? '';
  $('stage').classList.toggle('offline', !state.online);
  if (state.screens.length > 1) timer = setTimeout(next, secs * 1000);
}

function next() {
  state.index = (state.index + 1) % state.screens.length;
  show();
}

async function refresh() {
  const snap = await loadSnapshot(client, storage);
  if (snap) {
    state.courses = pickCourses(snap.rows, localISO());
    state.config = snap.config;
    state.online = snap.online;
    state.loaded = true;
  } else if (!state.loaded) {
    state.online = false;
  }
  rebuild();
  if (firstLoad || state.screens.length === 1) {
    firstLoad = false;
    show();
  } else {
    $('stage').classList.toggle('offline', !state.online);
  }
}

async function refreshLoop() {
  await refresh();
  setTimeout(refreshLoop, state.loaded ? REFRESH_MS : RETRY_MS);
}

function measureFps(ms = 3000) {
  return new Promise((resolve) => {
    let frames = 0;
    const t0 = performance.now();
    requestAnimationFrame(function f(t) {
      frames++;
      if (t - t0 < ms) requestAnimationFrame(f);
      else resolve(frames / ((t - t0) / 1000));
    });
  });
}

async function watchPerf() {
  if (root.classList.contains('lite')) return;
  const fps = await measureFps();
  if (fps < 24) root.classList.add('lite');
  else setTimeout(watchPerf, 2 * 60 * 1000);
}

fit();
addEventListener('resize', fit);
tick();
setInterval(tick, 10000);
refreshLoop();
setTimeout(watchPerf, 4000);
window.__tv = { refresh, state };
```
